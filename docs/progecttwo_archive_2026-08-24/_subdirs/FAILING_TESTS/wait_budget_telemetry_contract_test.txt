              '  /// Places the world origin at `distanceMeters` ahead of the camera\'s\n'
              '  /// current optical axis, captures the camera\'s bearing as worldYaw.\n'
              '  /// Returns the dictionary that becomes the Dart-side response.\n'
              '  /// The phone-orientation classification (portrait vs landscape) is\n'
              '  /// done on the Dart side by `PhoneOrientationClassifier` so the\n'
              '  /// algorithm stays cross-platform.\n'
              '  ///\n'
              '  /// Returns nil when ARKit\'s tracking state hasn\'t reached `.normal`\n'
              '  /// — the first few ARFrames typically arrive under `.notAvailable`\n'
              '  /// / `.limited` with an identity-ish transform, and locking against\n'
              '  /// one of those produces a bogus origin / worldYaw. The Dart-side\n'
              '  /// retry loop in `CaptureSession._lockOriginWhenReady` keeps\n'
              '  /// polling every 100 ms until tracking stabilises.\n'
              '  private func lockOrigin(distanceMeters: Float) -> [String: Any]? {\n'
              '    guard let frame = arSession?.currentFrame else {\n'
              '      NSLog("[OfficialAetherARKit] lockOrigin: no currentFrame yet")\n'
              '      return nil\n'
              '    }\n'
              '    switch frame.camera.trackingState {\n'
              '    case .normal:\n'
              '      break\n'
              '    case .limited(let reason):\n'
              '      NSLog("[OfficialAetherARKit] lockOrigin: tracking is .limited(\\(reason)) — retrying")\n'
              '      return nil\n'
              '    case .notAvailable:\n'
              '      NSLog("[OfficialAetherARKit] lockOrigin: tracking .notAvailable — retrying")\n'
              '      return nil\n'
              '    @unknown default:\n'
              '      NSLog("[OfficialAetherARKit] lockOrigin: unknown trackingState — retrying")\n'
              '      return nil\n'
              '    }\n'
              '    let t = frame.camera.transform\n'
              '    let camPos = simd_float3(t.columns.3.x, t.columns.3.y, t.columns.3.z)\n'
              '    // Forward = camera\'s optical axis (-Z column of the camera\n'
              '    // transform). Lock targets whatever\'s at the center of the screen.\n'
              '    // Y component is preserved on purpose: lock-time tilt is what makes\n'
              '    // "shoot the object from 45° above → dome shows the +45° cell"\n'
              '    // work without any extra orientation math.\n'
              '    let forward = -simd_float3(t.columns.2.x, t.columns.2.y, t.columns.2.z)\n'
              '\n'
              '    // Pick the lock POSITION via a tiered raycast strategy:\n'
              '    //\n'
              '    //   1. `.estimatedPlane / .any` — ARKit fits a virtual plane to\n'
              '    //      nearby feature points along the gaze direction, regardless\n'
              '    //      of orientation. Hits upright surfaces (a paper bag\'s side,\n'
              '    //      a chair\'s back, a figurine) where no detected horizontal\n'
              '    //      plane exists. This is what fixes the "depth wrong" symptom\n'
              '    //      where `.existingPlaneInfinite, .horizontal` silently\n'
              '    //      sailed past the subject and hit the floor 0.47 m in front\n'
              '    //      of the user instead of the actual subject.\n'
              '    //   2. `.existingPlaneInfinite, .horizontal` — fallback for the\n'
              '    //      case where ARKit hasn\'t accumulated enough feature points\n'
              '    //      to estimate a plane yet, but has detected a real horizontal\n'
              '    //      surface. Same behavior as before.\n'
              '    //   3. forward × distanceMeters — final mid-air fallback if\n'
              '    //      neither raycast lands.\n'
              '    //\n'
              '    // Cap=2.5 m: subjects beyond that are usually mis-aimed (raycast\n'
              '    // sails past intended subject); fall back to forward × distance\n'
              '    // so the anchor stays close enough to ARKit\'s feature cloud for\n'
              '    // stable tracking.\n'
              '    let subjectAnchorMaxRange: Float = 2.5\n'
              '    var origin: simd_float3\n'
              '    var positionSource: String\n'
              '    if let session = arSession {\n'
              '      var hits: [ARRaycastResult] = []\n'
              '      var raycastSource: String = ""\n'
              '      if #available(iOS 13.0, *) {\n'
              '        let estimateQuery = ARRaycastQuery(\n'
              '          origin: camPos,\n'
              '          direction: simd_normalize(forward),\n'
              '          allowing: .estimatedPlane,\n'
              '          alignment: .any\n'
              '        )\n'
              '        hits = session.raycast(estimateQuery)\n'
              '        if !hits.isEmpty { raycastSource = "estimated plane" }\n'
              '      }\n'
              '      if hits.isEmpty {\n'
              '        let infQuery = ARRaycastQuery(\n'
              '          origin: camPos,\n'
              '          direction: simd_normalize(forward),\n'
              '          allowing: .existingPlaneInfinite,\n'
              '          alignment: .horizontal\n'
              '        )\n'
              '        hits = session.raycast(infQuery)\n'
              '        if !hits.isEmpty { raycastSource = "existing horizontal plane" }\n'
              '      }\n'
              '      if let hit = hits.first {\n'
              '        let hitPos = simd_float3(\n'
              '          hit.worldTransform.columns.3.x,\n'
              '          hit.worldTransform.columns.3.y,\n'
              '          hit.worldTransform.columns.3.z\n'
              '        )\n'
              '        let hitDistance = simd_distance(camPos, hitPos)\n'
              '        if hitDistance <= subjectAnchorMaxRange {\n'
              '          origin = hitPos\n'
              '          positionSource = "\\(raycastSource) (\\(String(format: "%.2f", hitDistance)) m)"\n'
              '        } else {\n'
              '          origin = camPos + simd_normalize(forward) * distanceMeters\n'
              '          positionSource = "forward fallback (\\(raycastSource) hit \\(String(format: "%.2f", hitDistance)) m > cap \\(subjectAnchorMaxRange) m)"\n'
              '        }\n'
              '      } else {\n'
              '        origin = camPos + simd_normalize(forward) * distanceMeters\n'
              '        positionSource = "forward fallback (no raycast hit)"\n'
              '      }\n'
              '    } else {\n'
              '      origin = camPos + simd_normalize(forward) * distanceMeters\n'
              '      positionSource = "forward fallback (no session)"\n'
              '    }\n'
              '\n'
              '    // Drop any previous subject anchor — a fresh lock means we\'re\n'
              '    // starting over.\n'
              '    if let oldAnchor = worldSubjectAnchor, let session = arSession {\n'
              '      session.remove(anchor: oldAnchor)\n'
              '      worldSubjectAnchor = nil\n'
              '    }\n'
              '    lockTimeAnchorTransform = nil\n'
              '    lastLoggedAnchorTransform = nil\n'
              '    lastAnchorSeverity = nil\n'
              '    lastAnchorV2Transform = nil\n'
              '    lastAnchorV2LogTime = 0\n'
              '\n'
              '    // Install a single named ARAnchor at the chosen origin. ARKit\n'
              '    // tracks its transform across world-frame re-alignments;\n'
              '    // broadcast() re-reads it every frame to update worldOrigin in\n'
              '    // lock-step. WWDC 2018 §610 + Polycam polyform pattern — the\n'
              '    // canonical ARKit-correct way to pin a real-world point.\n'
              '    if let session = arSession {\n'
              '      var transform = matrix_identity_float4x4\n'
              '      transform.columns.3 = simd_float4(origin.x, origin.y, origin.z, 1)\n'

              '                            transform: transform)\n'

              '      worldSubjectAnchor = anchor\n'
              '      lockTimeAnchorTransform = transform\n'
              '      lastLoggedAnchorTransform = nil\n'
              '      lastAnchorSeverity = nil\n'
              '      lastAnchorV2Transform = nil\n'
              '    }\n'
              '\n'
              '    // worldYaw = "camera\'s relative bearing at lock". Subsequent\n'
              '    // frames\' azimuth subtracts this so the dome\'s az=0 ↔ lock pose.\n'
              '    let relInitial = camPos - origin\n'
              '    let yaw = atan2(relInitial.z, relInitial.x)\n'
              '\n'
              '    worldOrigin = origin\n'
              '    worldYaw = yaw\n'
              '    lockTimeOrigin = origin\n'
              '    lastDriftLogTime = 0  // force first drift log on next broadcast\n'
              '    lastAnchorV2LogTime = 0\n'

              '      trackingStateName: "normal",\n'
              '      currentTransform: lockTimeAnchorTransform,\n'
              '      anchorState: "locked"\n'

              '    if #available(iOS 16.0, *) {\n'
              '      restoreContinuousExposureFocus(reason: "subject lock")\n'
              '    }\n'
              '\n'
              '    return [\n'
              '      "originX": origin.x,\n'
              '      "originY": origin.y,\n'
              '      "originZ": origin.z,\n'
              '      "worldYaw": yaw,\n'
              '    ]\n'
              '  }\n'

              '  // MARK: Save current frame as JPEG (Plan G W2 photos-on-disk arch)\n'
              '  //\n'
              '  // Called by Dart\'s CaptureSession when a dome cell admits a frame:\n'
              '  // encode the most-recent ARFrame\'s pixel buffer to `<photosDir>/\n'
              '  // cell_<i>_slot_<j>.jpg` and write per-photo metadata JSON to a\n'
              '  // sibling `.json`. Eviction overwrites both files at the same path.\n'
              '  //\n'
              '  // Why on a dedicated background queue: the ImageIO encode of a 4K\n'
              '  // BGRA pixel buffer to JPEG q=0.9 takes ~30-50 ms on iPhone 14 Pro.\n'
              '  // Doing it on main thread would block the next pose tick. Doing it\n'
              '  // on the AR delegate\'s queue (also main) starves ARKit. The\n'
              '  // `jpegEncodeQueue` is dedicated and won\'t fight either.\n'
              '  //\n'
              '  // The snapshot is captured by VALUE (struct copy retains the\n'
              '  // CVPixelBuffer via ARC), so even if `lastFrameSnapshot` is\n'
              '  // overwritten by the next broadcast() during the encode, the closure\n'
              '  // holds the older snapshot until done. No race.\n'
              '  private func selectFrameSnapshot(\n'
              '    targetTimestamp: TimeInterval?,\n'
              '    maxTimestampDelta: TimeInterval\n'
              '  ) -> (\n'
              '    snapshot: LatestFrameSnapshot?,\n'
              '    delta: TimeInterval?,\n'
              '    errorMessage: String?\n'
              '  ) {\n'
              '    guard let targetTimestamp else {\n'
              '      return (lastFrameSnapshot, nil, nil)\n'
              '    }\n'
              '    guard !recentFrameSnapshots.isEmpty else {\n'
              '      return (nil, nil, "saveCurrentFrameAsJpeg: no ARFrame snapshots buffered")\n'
              '    }\n'
              '    var best: LatestFrameSnapshot?\n'
              '    var bestDelta = TimeInterval.greatestFiniteMagnitude\n'
              '    for snap in recentFrameSnapshots {\n'
              '      let delta = abs(snap.timestamp - targetTimestamp)\n'
              '      if delta < bestDelta {\n'
              '        best = snap\n'
              '        bestDelta = delta\n'
              '      }\n'
              '    }\n'
              '    if let best, bestDelta <= maxTimestampDelta {\n'
              '      return (best, bestDelta, nil)\n'
              '    }\n'
              '    return (\n'
              '      nil,\n'
              '      bestDelta,\n'
              '      String(\n'
              '        format: "saveCurrentFrameAsJpeg: nearest ARFrame is %.3fs from target %.6f, over max %.3fs",\n'
              '        bestDelta,\n'
              '        targetTimestamp,\n'
              '        maxTimestampDelta\n'
              '      )\n'
              '    )\n'
              '  }\n'

              '        }\n'

              '        try json.write(to: URL(fileURLWithPath: metadataPath))\n'
              '        // Streaming-SfM feed: attach an aspect-preserving grayscale of the\n'
              '        // SAME snapshot (so intrinsics/extrinsic below are frame-exact) for\n'
              '        // `aether_sfm_add_frame`. Best-effort — a nil gray just means the\n'
              '        // Dart side skips feeding this frame; the JPEG save already\n'
              '        // succeeded and is authoritative.\n'

              '          "t": snap.timestamp,\n'

              '  static func physFootprintMB() -> Double {\n'
              '    var info = task_vm_info_data_t()\n'
              '    var count = mach_msg_type_number_t(\n'
              '      MemoryLayout<task_vm_info_data_t>.size / MemoryLayout<integer_t>.size)\n'
              '    let kr = withUnsafeMutablePointer(to: &info) {\n'
              '      $0.withMemoryRebound(to: integer_t.self, capacity: Int(count)) { reb in\n'
              '        task_info(mach_task_self_, task_flavor_t(TASK_VM_INFO), reb, &count)\n'
              '      }\n'
              '    }\n'
              '    return (kr == KERN_SUCCESS) ? Double(info.phys_footprint) / 1_048_576.0 : -1.0\n'
              '  }\n'

              '  private static func computeScaleAlignPremetrics(\n'
              '    cameraTransform: simd_float4x4,\n'
              '    anchorsWorld: [[Float]]\n'
              '  ) -> ScaleAlignPremetrics {\n'
              '    let cameraPosition = SIMD3<Float>(\n'
              '      cameraTransform.columns.3.x,\n'
              '      cameraTransform.columns.3.y,\n'
              '      cameraTransform.columns.3.z\n'
              '    )\n'
              '    let cameraZAxisWorld = SIMD3<Float>(\n'
              '      cameraTransform.columns.2.x,\n'
              '      cameraTransform.columns.2.y,\n'
              '      cameraTransform.columns.2.z\n'
              '    )\n'
              '\n'
              '    var count = 0\n'
              '    var minDepth = Float.greatestFiniteMagnitude\n'
              '    var maxDepth = -Float.greatestFiniteMagnitude\n'
              '    for p in anchorsWorld {\n'
              '      if p.count < 3 { continue }\n'
              '      let worldPoint = SIMD3<Float>(p[0], p[1], p[2])\n'
              '      let delta = worldPoint - cameraPosition\n'
              '      // ARKit camera looks down local -Z. Positive scene depth is -cam.z.\n'
              '      let depth = -simd_dot(delta, cameraZAxisWorld)\n'
              '      if depth.isFinite && depth >= 0.10 && depth <= 6.0 {\n'
              '        count += 1\n'
              '        minDepth = min(minDepth, depth)\n'
              '        maxDepth = max(maxDepth, depth)\n'
              '      }\n'
              '    }\n'
              '\n'
              '    if count == 0 {\n'
              '      return ScaleAlignPremetrics(\n'
              '        anchorDepthCount: 0,\n'
              '        anchorDepthMinM: 0,\n'
              '        anchorDepthMaxM: 0,\n'
              '        anchorDepthSpanM: 0,\n'
              '        reliabilityPrior: 0\n'
              '      )\n'
              '    }\n'
              '\n'
              '    let span = max(0, maxDepth - minDepth)\n'
              '    let countScore = clamp01((Float(count) - 12.0) / 48.0)\n'
              '    let spanScore = clamp01((span - 0.08) / 0.42)\n'
              '    let reliability = clamp01(countScore * 0.45 + spanScore * 0.55)\n'
              '    return ScaleAlignPremetrics(\n'
              '      anchorDepthCount: count,\n'
              '      anchorDepthMinM: minDepth,\n'
              '      anchorDepthMaxM: maxDepth,\n'
              '      anchorDepthSpanM: span,\n'
              '      reliabilityPrior: reliability\n'
              '    )\n'
              '  }\n'
              '\n'
              '  private static func clamp01(_ x: Float) -> Float {\n'
              '    return min(1.0, max(0.0, x))\n'
              '  }\n'

              '  private func broadcast(frame: ARFrame) {\n'
              '    // Plan G W2 photos-on-disk arch (replaces the deleted AVAssetWriter\n'
              '    // pipeline 2026-05-16): keep a short timestamp-addressable snapshot\n'
              '    // ring so Dart can ask for the ARFrame that actually produced the\n'
              '    // accepted pose event, not whichever frame happens to be latest after\n'
              '    // MethodChannel round-trip latency.\n'
              '    //\n'
              '    // Why up-front (before payload assembly): saveCurrentFrameAsJpeg can\n'
              '    // fire from Dart any time after the corresponding pose event reaches\n'
              '    // the cell. We want the snapshot fresh by the time that round-trip\n'
              '    // completes (~50-100 ms later) — even if the rest of broadcast is\n'
              '    // still running, the snapshot is already valid.\n'
              '    let cameraTransform = frame.camera.transform\n'
              '    let cameraIntrinsics = frame.camera.intrinsics\n'
              '    let extrinsicArr: [Float] = [\n'
              '      cameraTransform.columns.0.x, cameraTransform.columns.0.y,\n'
              '      cameraTransform.columns.0.z, cameraTransform.columns.0.w,\n'
              '      cameraTransform.columns.1.x, cameraTransform.columns.1.y,\n'
              '      cameraTransform.columns.1.z, cameraTransform.columns.1.w,\n'
              '      cameraTransform.columns.2.x, cameraTransform.columns.2.y,\n'
              '      cameraTransform.columns.2.z, cameraTransform.columns.2.w,\n'
              '      cameraTransform.columns.3.x, cameraTransform.columns.3.y,\n'
              '      cameraTransform.columns.3.z, cameraTransform.columns.3.w,\n'
              '    ]\n'
              '    let intrinsicArr: [Float] = [\n'
              '      cameraIntrinsics.columns.0.x, // fx\n'
              '      cameraIntrinsics.columns.1.y, // fy\n'
              '      cameraIntrinsics.columns.2.x, // cx\n'
              '      cameraIntrinsics.columns.2.y, // cy\n'
              '    ]\n'
              '    let trackingStateName = Self.trackingStateString(frame.camera.trackingState)\n'
              '    let isTracking: Bool\n'
              '    switch frame.camera.trackingState {\n'
              '    case .normal: isTracking = true\n'
              '    default: isTracking = false\n'
              '    }\n'
              '    let pixelBuf = frame.capturedImage\n'
              '    let imgW = CVPixelBufferGetWidth(pixelBuf)\n'
              '    let imgH = CVPixelBufferGetHeight(pixelBuf)\n'
              '    var anchorsW: [[Float]] = []\n'
              '    var anchorIds: [UInt64] = []\n'
              '    if let raw = frame.rawFeaturePoints {\n'
              '      let n = raw.points.count\n'
              '      anchorsW.reserveCapacity(n)\n'
              '      anchorIds.reserveCapacity(n)\n'
              '      for i in 0..<n {\n'
              '        let p = raw.points[i]\n'
              '        anchorsW.append([p.x, p.y, p.z])\n'
              '        anchorIds.append(raw.identifiers[i])\n'
              '      }\n'
              '    }\n'
              '    let finiteAnchors = PWJSONSafety.finitePointPairs(\n'
              '      anchorsW,\n'
              '      identifiers: anchorIds\n'
              '    )\n'
              '    anchorsW = finiteAnchors.points\n'
              '    anchorIds = finiteAnchors.identifiers\n'
              '    let scaleAlignPremetrics = Self.computeScaleAlignPremetrics(\n'
              '      cameraTransform: cameraTransform,\n'
              '      anchorsWorld: anchorsW\n'
              '    )\n'
              '    let snapshot = LatestFrameSnapshot(\n'
              '      pixelBuffer: pixelBuf,\n'
              '      timestamp: frame.timestamp,\n'
              '      extrinsic: extrinsicArr,\n'
              '      intrinsicsFxFyCxCy: intrinsicArr,\n'
              '      imageW: imgW,\n'
              '      imageH: imgH,\n'
              '      trackingStateName: trackingStateName,\n'
              '      isTracking: isTracking,\n'
              '      anchorsWorld: anchorsW,\n'
              '      anchorIds: anchorIds,\n'
              '      scaleAlignPremetrics: scaleAlignPremetrics\n'
              '    )\n'
              '    lastFrameSnapshot = snapshot\n'
              '    recentFrameSnapshots.append(snapshot)\n'
              '    if recentFrameSnapshots.count > Self.maxRecentFrameSnapshots {\n'
              '      recentFrameSnapshots.removeFirst(\n'
              '        recentFrameSnapshots.count - Self.maxRecentFrameSnapshots\n'
              '      )\n'
              '    }\n'
              '\n'
              '    // ── Refresh worldOrigin from the subject anchor\'s latest transform.\n'
              '    // ARKit re-aligns its world frame continuously (limited→normal\n'
              '    // recovery, loop closure). Per WWDC 2018 §610 + Polycam polyform:\n'
              '    // an `ARAnchor`\'s transform is updated by ARKit in lock-step with\n'
              '    // those re-alignments, so reading it every frame keeps `worldOrigin`\n'
              '    // glued to the real-world point the user locked. We accept the\n'
              '    // update unconditionally — an earlier 0.5 m drift-rejection\n'
              '    // threshold got stuck rejecting forever once ARKit issued a real\n'
              '    // multi-meter correction (no recovery once `diff(old, new)` stayed\n'
              '    // above the cap; user-facing symptom: "白球还是会大跳去很远的地方").\n'
              '    if let myAnchor = worldSubjectAnchor,\n'
              '       let updatedAnchor = frame.anchors.first(\n'
              '         where: { $0.identifier == myAnchor.identifier }\n'
              '       ) {\n'
              '      worldSubjectAnchor = updatedAnchor\n'
              '      worldOrigin = simd_float3(\n'
              '        updatedAnchor.transform.columns.3.x,\n'
              '        updatedAnchor.transform.columns.3.y,\n'
              '        updatedAnchor.transform.columns.3.z\n'
              '      )\n'
              '    }\n'
              '\n'
              '    if frame.timestamp - lastAnchorV2LogTime > 1.0 {\n'
              '      logAnchorObservationV2(\n'
              '        frameTimestamp: frame.timestamp,\n'
              '        trackingStateName: trackingStateName,\n'
              '        currentTransform: worldSubjectAnchor?.transform\n'
              '      )\n'
              '    }\n'
              '\n'
              '    // [LIVE-CLOUD-DIAG V1] Observation only: persist the complete 6DoF\n'
              '    // subject-anchor correction. No value below is fed back to ARKit, the\n'
              '    // point cloud, or capture policy.\n'
              '    if let lockTransform = lockTimeAnchorTransform,\n'
              '       let currentTransform = worldSubjectAnchor?.transform {\n'
              '      let absolute = LiveCloudAnchorDiagnostics.delta(\n'
              '        lock: lockTransform,\n'
              '        current: currentTransform\n'
              '      )\n'
              '      let severity = LiveCloudAnchorDiagnostics.severity(\n'
              '        translationMeters: absolute.translationMeters\n'
              '      )\n'
              '      let severityChanged = severity != lastAnchorSeverity\n'
              '      if frame.timestamp - lastDriftLogTime > 1.0 || severityChanged {\n'
              '        let step = LiveCloudAnchorDiagnostics.delta(\n'
              '          lock: lastLoggedAnchorTransform ?? lockTransform,\n'
              '          current: currentTransform\n'
              '        )\n'
              '        OfficialPwNativeTelemetry.shared.log("arkit_anchor_delta_v1", [\n'
              '          "contract": "PW_LIVE_CLOUD_DIAG_V1_20260810",\n'
              '          "frame_t": frame.timestamp,\n'
              '          "dx_m": Double(absolute.translation.x),\n'
              '          "dy_m": Double(absolute.translation.y),\n'
              '          "dz_m": Double(absolute.translation.z),\n'
              '          "translation_m": absolute.translationMeters,\n'
              '          "qx": Double(absolute.rotation.imag.x),\n'
              '          "qy": Double(absolute.rotation.imag.y),\n'
              '          "qz": Double(absolute.rotation.imag.z),\n'
              '          "qw": Double(absolute.rotation.real),\n'
              '          "rotation_deg": absolute.rotationDegrees,\n'
              '          "step_translation_m": step.translationMeters,\n'
              '          "step_rotation_deg": step.rotationDegrees,\n'
              '          "severity": severity,\n'
              '          "tracking": trackingStateName,\n'
              '          "observation_only": true,\n'
              '        ])\n'
              '        NSLog(String(\n'
              '          format: "[OfficialAetherARKit] anchor drift: %.3f m / %.2f deg (%@)",\n'
              '          absolute.translationMeters,\n'
              '          absolute.rotationDegrees,\n'
              '          severity\n'
              '        ))\n'
              '        lastDriftLogTime = frame.timestamp\n'
              '        lastLoggedAnchorTransform = currentTransform\n'
              '        lastAnchorSeverity = severity\n'

              '      ))\n'
              '      qDiagWindowStart = frame.timestamp\n'
              '      qDiagFires = 0\n'
              '      qDiagSkips = 0\n'
              '      qDiagElapsedMsSum = 0\n'
              '      qDiagAttached = 0\n'
              '      qDiagPoseEvents = 0\n'
              '    }\n'
              '    // Include worldOrigin / worldYaw so the Dart side can do the\n'
              '    // (rel = camPos - origin) math without a round-trip back into\n'
              '    // ARKit. Always sent (zero before lock) so the schema is stable.\n'
              '    if let origin = worldOrigin {\n'

              '      payload["hasOrigin"] = false\n'
              '    }\n'
              '\n'
              '    poseStreamHandler.send(payload)\n'
              '  }\n'

              '  /// Pull the Y (luma) plane out of a YUV CVPixelBuffer and nearest-\n'
              '  /// neighbour downsample it to a 128×128 uint8 thumbnail.\n'
              '  ///\n'
              '  /// This is the new shape of what used to be `computeQuality` — the\n'
              '  /// Laplacian-variance + brightness + signature math has moved to\n'
              '  /// `lib/quality/quality_compute.dart` so it can run identically on\n'
              '  /// iOS / Android / Web / HarmonyOS without four separate ports.\n'
              '  /// Native still does the platform-specific plane extraction (only\n'
              '  /// way to get at the YUV buffer) but everything past that lives in\n'
              '  /// shared Dart.\n'
              '  ///\n'
              '  /// Cost: ~2-3 ms on iPhone 12 Pro (down from ~5-15 ms of the full\n'
              '  /// pre-Dart-port quality compute). Returns nil only when the pixel\n'
              '  /// buffer isn\'t one of the BiPlanar YUV variants ARKit normally\n'
              '  /// produces — caller treats nil as "skip this quality tick".\n'
              '  ///\n'
              '  /// Output is exactly 128×128 = 16384 bytes, row-major, top-left\n'
              '  /// origin, ready to ship across the platform channel as a single\n'
              '  /// FlutterStandardTypedData blob.\n'
              '  static func extractGray128(_ pixelBuffer: CVPixelBuffer) -> Data? {\n'
              '    return extractGray(pixelBuffer, targetSide: downsampleSide)\n'
              '  }\n'
              '\n'
              '  static func extractGray(\n'
              '    _ pixelBuffer: CVPixelBuffer,\n'
              '    targetSide: Int\n'
              '  ) -> Data? {\n'
              '    let format = CVPixelBufferGetPixelFormatType(pixelBuffer)\n'
              '    let isYUV =\n'
              '      format == kCVPixelFormatType_420YpCbCr8BiPlanarFullRange ||\n'
              '      format == kCVPixelFormatType_420YpCbCr8BiPlanarVideoRange\n'
              '    guard isYUV, targetSide > 0 else { return nil }\n'
              '\n'
              '    CVPixelBufferLockBaseAddress(pixelBuffer, .readOnly)\n'
              '    defer { CVPixelBufferUnlockBaseAddress(pixelBuffer, .readOnly) }\n'
              '\n'
              '    let width = CVPixelBufferGetWidthOfPlane(pixelBuffer, 0)\n'
              '    let height = CVPixelBufferGetHeightOfPlane(pixelBuffer, 0)\n'
              '    let rowStride = CVPixelBufferGetBytesPerRowOfPlane(pixelBuffer, 0)\n'
              '    guard let baseAddr = CVPixelBufferGetBaseAddressOfPlane(pixelBuffer, 0)\n'
              '    else { return nil }\n'
              '    let src = baseAddr.assumingMemoryBound(to: UInt8.self)\n'
              '\n'
              '    let tw = targetSide\n'
              '    let th = targetSide\n'
              '    var data = Data(count: tw * th)\n'
              '    data.withUnsafeMutableBytes { (raw: UnsafeMutableRawBufferPointer) in\n'
              '      let dst = raw.bindMemory(to: UInt8.self).baseAddress!\n'
              '      // Fixed-point bilinear-step-and-pick (nearest-neighbour). Match\n'
              '      // the math the previous Swift implementation used so the Dart\n'
              '      // port\'s results are byte-identical with the old wire format\n'
              '      // during the migration window.\n'
              '      let sxFixed = (width << 16) / tw\n'
              '      let syFixed = (height << 16) / th\n'
              '      for dy in 0..<th {\n'
              '        let srcY = (dy * syFixed) >> 16\n'
              '        let srcRowOffset = srcY * rowStride\n'
              '        let dstRowOffset = dy * tw\n'
              '        for dx in 0..<tw {\n'
              '          let srcX = (dx * sxFixed) >> 16\n'
              '          dst[dstRowOffset + dx] = src[srcRowOffset + srcX]\n'
              '        }\n'
              '      }\n'
              '    }\n'
              '    return data\n'
              '  }\n'
              '\n'
              '  /// Aspect-preserving variant of `extractGray` for the streaming-SfM feed:\n'
              '  /// scales the Y plane uniformly so the LONG edge equals `maxSide` (never\n'
              '  /// upscales). Uniform scale keeps fx/fy shrinking by the same factor, which\n'
              '  /// the SfM\'s single-focal SIMPLE_PINHOLE self-calibration requires — the\n'
              '  /// square `extractGray` squash must NOT be used for SfM input.\n'
              '  /// Output is row-major top-down 8-bit gray (CGImage convention), exactly\n'
              '  /// what `aether_sfm_add_frame` consumes.\n'
              '  static func extractGrayAspect(\n'
              '    _ pixelBuffer: CVPixelBuffer,\n'
              '    maxSide: Int\n'
              '  ) -> (data: Data, width: Int, height: Int)? {\n'
              '    let format = CVPixelBufferGetPixelFormatType(pixelBuffer)\n'
              '    let isYUV =\n'
              '      format == kCVPixelFormatType_420YpCbCr8BiPlanarFullRange ||\n'
              '      format == kCVPixelFormatType_420YpCbCr8BiPlanarVideoRange\n'
              '    guard isYUV, maxSide > 0 else { return nil }\n'
              '\n'
              '    CVPixelBufferLockBaseAddress(pixelBuffer, .readOnly)\n'
              '    defer { CVPixelBufferUnlockBaseAddress(pixelBuffer, .readOnly) }\n'
              '\n'
              '    let width = CVPixelBufferGetWidthOfPlane(pixelBuffer, 0)\n'
              '    let height = CVPixelBufferGetHeightOfPlane(pixelBuffer, 0)\n'
              '    guard width > 0, height > 0 else { return nil }\n'
              '    let rowStride = CVPixelBufferGetBytesPerRowOfPlane(pixelBuffer, 0)\n'
              '    guard let baseAddr = CVPixelBufferGetBaseAddressOfPlane(pixelBuffer, 0)\n'
              '    else { return nil }\n'
              '    let src = baseAddr.assumingMemoryBound(to: UInt8.self)\n'
              '\n'
              '    let longEdge = max(width, height)\n'
              '    // Never upscale: uniform factor <= 1.\n'
              '    let scaleNum = min(maxSide, longEdge)\n'
              '    let tw = max(1, width * scaleNum / longEdge)\n'
              '    let th = max(1, height * scaleNum / longEdge)\n'
              '    var data = Data(count: tw * th)\n'
              '    data.withUnsafeMutableBytes { (raw: UnsafeMutableRawBufferPointer) in\n'
              '      let dst = raw.bindMemory(to: UInt8.self).baseAddress!\n'
              '      // Box-average (area) downsample — replaces the old nearest-neighbor\n'
              '      // step-and-pick (2026-07-08). At the 3840→2000 live tier (~1.9× down)\n'
              '      // nearest-neighbor kept only 1 of every ~3.7 source pixels and discarded\n'
              '      // the rest, which aliased and — the real damage for photogrammetry —\n'
              '      // erased the weak-texture gradients (bedsheets, flat detail) the SIFT DoG\n'
              '      // detector needs, so those keypoints vanished and the live cloud thinned\n'
              '      // (28k vs 49k at full-res). Averaging each destination pixel over its full\n'
              '      // source block preserves those gradients so more real keypoints survive\n'
              '      // the downscale. The dst blocks tile the source plane exactly, so this is\n'
              '      // ONE pass over the source (~a few ms; detector is ~468 ms/frame — free).\n'
              '      // COLMAP downsamples with a proper filter for the same reason; the\n'
              '      // nearest-neighbor pick was the bug. Reduces to identity when tw==width\n'
              '      // (no-downscale research tier), block size 1.\n'
              '      for dy in 0..<th {\n'
              '        let sy0 = dy * height / th\n'
              '        var sy1 = (dy + 1) * height / th\n'
              '        if sy1 <= sy0 { sy1 = sy0 + 1 }\n'
              '        let dstRowOffset = dy * tw\n'
              '        for dx in 0..<tw {\n'
              '          let sx0 = dx * width / tw\n'
              '          var sx1 = (dx + 1) * width / tw\n'
              '          if sx1 <= sx0 { sx1 = sx0 + 1 }\n'
              '          var sum = 0\n'
              '          var cnt = 0\n'
              '          var sy = sy0\n'
              '          while sy < sy1 {\n'
              '            let rowOff = sy * rowStride\n'
              '            var sx = sx0\n'
              '            while sx < sx1 {\n'
              '              sum += Int(src[rowOff + sx])\n'
              '              cnt += 1\n'
              '              sx += 1\n'
              '            }\n'
              '            sy += 1\n'
              '          }\n'
              '          dst[dstRowOffset + dx] = UInt8(sum / cnt)\n'
              '        }\n'
              '      }\n'
              '\n'
              '      // Photogrammetry preflight (2026-07-05, "微暗是常态"): percentile\n'
              '      // contrast stretch so DIM indoor captures — the normal case — feed\n'
              '      // the SIFT DoG detector at full contrast instead of starving it.\n'
              '      // 2%..98% of the histogram maps to 0..255; bright scenes are near\n'
              '      // identity (p2≈0, p98≈255), gain is capped at 8× so near-black\n'
              '      // noise is never amplified into fake texture. One extra pass over\n'
              '      // the buffer (~5 ms at 4K) — same normalization every frame, so the\n'
              '      // shared-camera / consistent-appearance premise holds.\n'
              '      let n = tw * th\n'
              '      var hist = [Int](repeating: 0, count: 256)\n'
              '      for i in 0..<n { hist[Int(dst[i])] += 1 }\n'
              '      let lowCount = n / 50        // 2%\n'
              '      let highCount = n - n / 50   // 98%\n'
              '      var acc = 0\n'
              '      var p2 = 0\n'
              '      var p98 = 255\n'
              '      for v in 0..<256 {\n'
              '        acc += hist[v]\n'
              '        if acc <= lowCount { p2 = v }\n'
              '        if acc < highCount { p98 = v }\n'
              '      }\n'
              '      let span = max(32, p98 - p2)  // cap gain at ~8×\n'
              '      if p2 > 0 || span < 250 {\n'
              '        var lut = [UInt8](repeating: 0, count: 256)\n'
              '        for v in 0..<256 {\n'
              '          let stretched = (v - p2) * 255 / span\n'
              '          lut[v] = UInt8(min(255, max(0, stretched)))\n'
              '        }\n'
              '        for i in 0..<n { dst[i] = lut[Int(dst[i])] }\n'
              '      }\n'
              '    }\n'
              '    return (data, tw, th)\n'
              '  }\n'
              '\n'
              '  /// Build a small, color-sampled preview point payload from ARKit\'s\n'
              '  /// official VIO feature cloud. This mirrors the RealityScan/Polycam\n'
              '  /// capture-time idea at the executor boundary: native only exposes\n'
              '  /// raw world-space points + sampled RGB; Dart performs multi-scale\n'
              '  /// voxel hashing and UI policy.\n'
              '  static func makePreviewPointPayload(\n'
              '    frame: ARFrame,\n'
              '    maxPoints: Int\n'
              '  ) -> [String: Any] {\n'
              '    guard maxPoints > 0, let raw = frame.rawFeaturePoints else {\n'
              '      return [:]\n'
              '    }\n'
              '    let rawCount = raw.points.count\n'
              '    guard rawCount > 0 else { return [:] }\n'
              '\n'
              '    let pixelBuffer = frame.capturedImage\n'
              '    let width = CVPixelBufferGetWidth(pixelBuffer)\n'
              '    let height = CVPixelBufferGetHeight(pixelBuffer)\n'
              '    guard width > 0, height > 0 else { return [:] }\n'
              '\n'
              '    let step = max(1, rawCount / maxPoints)\n'
              '    let viewport = CGSize(width: width, height: height)\n'
              '    var xyz: [Float] = []\n'
              '    var rgb: [Int] = []\n'
              '    var confidence: [Float] = []\n'
              '    xyz.reserveCapacity(min(maxPoints, rawCount) * 3)\n'
              '    rgb.reserveCapacity(min(maxPoints, rawCount) * 3)\n'
              '    confidence.reserveCapacity(min(maxPoints, rawCount))\n'
              '\n'
              '    CVPixelBufferLockBaseAddress(pixelBuffer, .readOnly)\n'
              '    defer { CVPixelBufferUnlockBaseAddress(pixelBuffer, .readOnly) }\n'
              '\n'
              '    for i in Swift.stride(from: 0, to: rawCount, by: step) {\n'
              '      if confidence.count >= maxPoints { break }\n'
              '      let p = raw.points[i]\n'
              '      let projected = frame.camera.projectPoint(\n'
              '        p,\n'
              '        orientation: .landscapeRight,\n'
              '        viewportSize: viewport\n'
              '      )\n'
              '      let x = Int(projected.x.rounded())\n'
              '      let y = Int(projected.y.rounded())\n'
              '      guard x >= 0, y >= 0, x < width, y < height else { continue }\n'
              '      guard let color = sampleYuvRgbLocked(pixelBuffer, x: x, y: y) else {\n'
              '        continue\n'
              '      }\n'
              '      xyz.append(p.x)\n'
              '      xyz.append(p.y)\n'
              '      xyz.append(p.z)\n'
              '      rgb.append(Int(color.r))\n'
              '      rgb.append(Int(color.g))\n'
              '      rgb.append(Int(color.b))\n'
              '      confidence.append(1.0)\n'
              '    }\n'
              '\n'
              '    if confidence.isEmpty { return [:] }\n'
              '    return [\n'
              '      "previewPointXYZ": xyz,\n'
              '      "previewPointRGB": rgb,\n'
              '      "previewPointConfidence": confidence,\n'
              '      "previewPointSource": "arkit_rawFeaturePoints_voxel_preview",\n'
              '    ]\n'
              '  }\n'

              '  private static func sampleYuvRgbLocked(\n'
              '    _ pixelBuffer: CVPixelBuffer,\n'
              '    x: Int,\n'
              '    y: Int\n'
              '  ) -> (r: UInt8, g: UInt8, b: UInt8)? {\n'
              '    let format = CVPixelBufferGetPixelFormatType(pixelBuffer)\n'
              '    let isYUV =\n'
              '      format == kCVPixelFormatType_420YpCbCr8BiPlanarFullRange ||\n'
              '      format == kCVPixelFormatType_420YpCbCr8BiPlanarVideoRange\n'
              '    guard isYUV else { return nil }\n'
              '\n'
              '    let yWidth = CVPixelBufferGetWidthOfPlane(pixelBuffer, 0)\n'
              '    let yHeight = CVPixelBufferGetHeightOfPlane(pixelBuffer, 0)\n'
              '    let uvWidth = CVPixelBufferGetWidthOfPlane(pixelBuffer, 1)\n'
              '    let uvHeight = CVPixelBufferGetHeightOfPlane(pixelBuffer, 1)\n'
              '    guard x >= 0, y >= 0, x < yWidth, y < yHeight else { return nil }\n'
              '\n'
              '    guard\n'
              '      let yBase = CVPixelBufferGetBaseAddressOfPlane(pixelBuffer, 0),\n'
              '      let uvBase = CVPixelBufferGetBaseAddressOfPlane(pixelBuffer, 1)\n'
              '    else {\n'
              '      return nil\n'
              '    }\n'
              '\n'
              '    let yStride = CVPixelBufferGetBytesPerRowOfPlane(pixelBuffer, 0)\n'
              '    let uvStride = CVPixelBufferGetBytesPerRowOfPlane(pixelBuffer, 1)\n'
              '    let yPtr = yBase.assumingMemoryBound(to: UInt8.self)\n'
              '    let uvPtr = uvBase.assumingMemoryBound(to: UInt8.self)\n'
              '\n'
              '    let uvX = min(max(x / 2, 0), max(uvWidth - 1, 0))\n'
              '    let uvY = min(max(y / 2, 0), max(uvHeight - 1, 0))\n'
              '    let yValue = Float(yPtr[y * yStride + x])\n'
              '    let uvIndex = uvY * uvStride + uvX * 2\n'
              '    let cb = Float(uvPtr[uvIndex]) - 128.0\n'
              '    let cr = Float(uvPtr[uvIndex + 1]) - 128.0\n'
              '\n'
              '    let r = yValue + 1.402 * cr\n'
              '    let g = yValue - 0.344136 * cb - 0.714136 * cr\n'
              '    let b = yValue + 1.772 * cb\n'

              '      r: clampRgb(r),\n'

              '      b: clampRgb(b)\n'

              '  }\n'

              '/// Capture-only adaptive point budget.\n'
              '///\n'
              '/// The full progressive cloud remains resident. Lower tiers are strict\n'
              '/// prefixes of the Potree-style octree order prepared by Dart, so a tier\n'
              '/// change never reshuffles surviving points. Decisions use a smoothed render\n'
              '/// cadence, thermal pressure, two-second decision windows, and a long upgrade\n'
              '/// hysteresis; the final PLY and Review viewer never pass through this class.\n'
              'private final class CapturePointCloudLodController {\n'
              '  private let fractions = [1.0, 0.67, 0.40, 0.25]\n'
              '  private var tier = 0\n'
              '  private var lastFrameTime: TimeInterval?\n'
              '  private var smoothedFrameSeconds = 1.0 / 30.0\n'
              '  private var lastDecisionTime: TimeInterval = 0\n'
              '  private var healthySince: TimeInterval?\n'
              '\n'
              '  func reset(totalPoints: Int) -> Int {\n'
              '    lastFrameTime = nil\n'
              '    healthySince = nil\n'
              '    // Small clouds are cheaper than rebuilding a lower tier.\n'
              '    if totalPoints <= 24_000 { tier = 0 }\n'
              '    return renderCount(totalPoints: totalPoints)\n'
              '  }\n'
              '\n'
              '  func renderCount(totalPoints: Int) -> Int {\n'
              '    guard totalPoints > 0 else { return 0 }\n'
              '    if totalPoints <= 24_000 { return totalPoints }\n'
              '    let fraction = fractions[min(tier, fractions.count - 1)]\n'
              '    return min(totalPoints, max(12_000, Int((Double(totalPoints) * fraction).rounded(.up))))\n'
              '  }\n'
              '\n'
              '  /// Returns a new prefix size only when the stable tier actually changes.\n'
              '  func observeFrame(time: TimeInterval, totalPoints: Int) -> Int? {\n'
              '    defer { lastFrameTime = time }\n'
              '    guard totalPoints > 24_000, let previous = lastFrameTime else { return nil }\n'
              '    let dt = time - previous\n'
              '    guard dt > 0, dt < 0.25 else { return nil }\n'
              '    smoothedFrameSeconds = smoothedFrameSeconds * 0.92 + dt * 0.08\n'
              '    guard time - lastDecisionTime >= 2.0 else { return nil }\n'
              '    lastDecisionTime = time\n'
              '\n'
              '    let fps = 1.0 / max(smoothedFrameSeconds, 1.0 / 120.0)\n'
              '    let thermal = ProcessInfo.processInfo.thermalState\n'
              '    let oldTier = tier\n'
              '\n'
              '    switch thermal {\n'
              '    case .critical:\n'
              '      tier = max(tier, 3)\n'
              '      healthySince = nil\n'
              '    case .serious:\n'
              '      tier = max(tier, 2)\n'
              '      healthySince = nil\n'
              '    case .nominal, .fair:\n'
              '      if fps < 24.0 {\n'
              '        tier = min(tier + 1, fractions.count - 1)\n'
              '        healthySince = nil\n'
              '      } else if fps >= 28.5 {\n'
              '        if healthySince == nil { healthySince = time }\n'
              '        // Upgrades need a long healthy window; downgrades react in one window.\n'
              '        if tier > 0, time - (healthySince ?? time) >= 8.0 {\n'
              '          tier -= 1\n'
              '          healthySince = time\n'
              '        }\n'
              '      } else {\n'
              '        healthySince = nil\n'
              '      }\n'
              '    @unknown default:\n'
              '      healthySince = nil\n'
              '    }\n'
              '\n'
              '    guard tier != oldTier else { return nil }\n'
              '    let count = renderCount(totalPoints: totalPoints)\n'
              '    OfficialPwNativeTelemetry.shared.log("capture_point_lod", [\n'
              '      "from_tier": oldTier,\n'
              '      "to_tier": tier,\n'
              '      "render_points": count,\n'
              '      "source_points": totalPoints,\n'
              '      "fps_ewma": (fps * 10).rounded() / 10,\n'
              '      "thermal": thermal.rawValue,\n'
              '    ])\n'
              '    return count\n'
              '  }\n'

              '  /// Dumb state→colour map(policy lives in Dart, photo_card_state.dart):\n'
              '  /// 0 黑=SfM 未处理 / 1 白=已注册 / 2 红=断联 / 3 黄=已注册但低视差。\n'

              '    switch state {\n'
              '    case 1: return .white\n'

              '    }\n'
              '  }\n'

              '  // ── Capture sparse cloud: full resident data + stable dynamic LOD ─────────\n'

              '  // the full display copy resident and draws one SceneKit point batch. Under\n'

              '  private var pointCloudNode: SCNNode?\n'
              '  private var fullPointCloudXyz: [Float] = []\n'
              '  private var fullPointCloudRgb: [UInt8] = []\n'
              '  private let pointCloudLod = CapturePointCloudLodController()\n'
              '  private var renderedPointCount = 0\n'
              '  private var liveCloudRenderApplySequence = 0\n'
              '\n'
              '  /// Render-loop tick: apply the latest Dart-pushed cloud when it changed.\n'

              '  private func updateFeaturePointOverlay(at time: TimeInterval) {\n'
              '    if !OfficialAetherARKitPlugin.featurePointsVisible {\n'

              '\n'
              '  private func rebuildPointCloud(renderCount: Int) {\n'
              '    let available = min(fullPointCloudXyz.count / 3, fullPointCloudRgb.count / 3)\n'
              '    let n = min(max(0, renderCount), available)\n'
              '    guard n > 0 else {\n'
              '      pointCloudNode?.removeFromParentNode()\n'
              '      pointCloudNode = nil\n'
              '      return\n'
              '    }\n'
              '    // Stable nested LOD: every lower tier is a prefix of the higher tier.\n'
              '    let xyz = Array(fullPointCloudXyz.prefix(renderCount * 3))\n'
              '    let rgb = Array(fullPointCloudRgb.prefix(renderCount * 3))\n'
              '    var verts: [SCNVector3] = []\n'
              '    var colors: [SIMD4<Float>] = []\n'
              '    verts.reserveCapacity(n)\n'
              '    colors.reserveCapacity(n)\n'
              '    for i in 0..<n {\n'
              '      verts.append(SCNVector3(xyz[i * 3], xyz[i * 3 + 1], xyz[i * 3 + 2]))\n'
              '      colors.append(SIMD4<Float>(\n'
              '        Float(rgb[i * 3]) / 255.0,\n'
              '        Float(rgb[i * 3 + 1]) / 255.0,\n'
              '        Float(rgb[i * 3 + 2]) / 255.0,\n'
              '        1.0))\n'
              '    }\n'
              '    let vSource = SCNGeometrySource(vertices: verts)\n'
              '    let cData = colors.withUnsafeBytes { Data($0) }\n'
              '    let cSource = SCNGeometrySource(\n'
              '      data: cData, semantic: .color, vectorCount: colors.count,\n'
              '      usesFloatComponents: true, componentsPerVector: 4,\n'
              '      bytesPerComponent: MemoryLayout<Float>.size, dataOffset: 0,\n'
              '      dataStride: MemoryLayout<SIMD4<Float>>.stride)\n'
              '    let element = SCNGeometryElement(\n'
              '      indices: (0..<verts.count).map { Int32($0) }, primitiveType: .point)\n'
              '    // [SPLAT-RADIUS 2026-07-28] 实验臂:近处点径上限。SceneKit 的 pointSize\n'
              '    // 是**世界单位**,min/max ScreenSpaceRadius 把投影后的像素半径钳在区间内\n'
              '    // —— 即"1/深度 透视自适应 + 钳制",与 Potree 的出货配方同构\n'
              '    // (`pointSize = size * spacing * projFactor` 后 clamp)。\n'
              '    //\n'
              '    // 病灶:上限 6px 把近处点焊死在小圆点,表面永远糊不成片。参照:Potree\n'
              '    // 出货推荐 clamp(2, 50);RS 手机端截图取证近端实测 7-11px(我们上限\n'
              '    // 6px 连它的下沿都够不到)。⇒ 放开上限是唯一"一行见效"的动作。\n'
              '    //\n'
              '    // 纯渲染:点数据一个字节不动,几何/交付/导出全不受影响。\n'
              '    // 档案:project_pocketworld_pointcloud_rendering_audit(记忆库)。\n'
              '    element.pointSize = 6\n'
              '    element.minimumPointScreenSpaceRadius = 2\n'
              '    element.maximumPointScreenSpaceRadius =\n'
              '      OfficialAetherARKitPlugin.arSplatMaxScreenRadius\n'
              '    let geo = SCNGeometry(sources: [vSource, cSource], elements: [element])\n'
              '    let mat = SCNMaterial()\n'
              '    mat.lightingModel = .constant\n'
              '    mat.diffuse.contents = UIColor.white      // × per-vertex colour\n'
              '    mat.writesToDepthBuffer = false\n'
              '    mat.readsFromDepthBuffer = false\n'
              '    geo.materials = [mat]\n'
              '    if let node = pointCloudNode {\n'
              '      node.geometry = geo\n'
              '    } else {\n'
              '      let node = SCNNode(geometry: geo)\n'
              '      node.renderingOrder = -10               // render under the photo cards\n'
              '      arscnView.scene.rootNode.addChildNode(node)\n'
              '      pointCloudNode = node\n'
              '    }\n'
              '  }\n'

              '  // MARK: ARSCNViewDelegate\n'
              '\n'
              '  /// Fires when ARKit adds an anchor to the session. ARSCNView creates\n'
              '  /// the parent SCNNode for us; we attach a child sphere if this is OUR\n'
              '  /// subject anchor (filtered by name to ignore plane anchors that\n'
              '  /// `planeDetection = [.horizontal]` adds automatically).\n'
              '  func renderer(_ renderer: SCNSceneRenderer, didAdd node: SCNNode, for anchor: ARAnchor) {\n'
              '    // Subject-origin anchor: no visible marker.\n'
              '    // Photo-card anchors: build a CUSTOM QUAD whose 4 corners are the unprojected\n'
              '    // viewport corners (so it pixel-aligns with the live view at capture). The\n'
              '    // texture is normalized to upright portrait (uprightPortrait) then aspect-\n'
              '    // filled with top-left-origin UVs. World-anchored, it "peels off the lens"\n'
              '    // as the camera moves. No orientation/size tuning.\n'
              '    guard let name = anchor.name, name.hasPrefix("official_photo_card_") else { return }\n'
              '    NSLog("[PHOTOCARD] renderer didAdd %@", name)\n'
              '    guard let spec = OfficialAetherARKitPlugin.photoCardSpecs[name] else {\n'
              '      NSLog("[PHOTOCARD] renderer: spec MISSING for %@", name)\n'
              '      return\n'
              '    }\n'
              '    // LOW-RES AR texture (RS-style memory saver). Decode a small thumbnail\n'
              '    // DIRECTLY from the 4K JPEG via ImageIO — it never decodes the full frame,\n'
              '    // so each floating card holds a ~1 MB texture instead of ~33 MB and hundreds\n'
              '    // of cards won\'t OOM. The album + DA3/SfM still read the full-res 4K JPEG on\n'
              '    // disk; only the AR card is downscaled. kCGImageSource…WithTransform bakes the\n'
              '    // EXIF orientation → upright portrait (replaces the manual uprightPortrait).\n'
              '    let thumbOpts: [CFString: Any] = [\n'
              '      kCGImageSourceCreateThumbnailFromImageAlways: true,\n'
              '      kCGImageSourceCreateThumbnailWithTransform: true,\n'
              '      kCGImageSourceThumbnailMaxPixelSize: OfficialAetherARKitPlugin.photoCardThumbMaxPx,\n'
              '    ]\n'
              '    guard let imgSrc = CGImageSourceCreateWithURL(\n'
              '            URL(fileURLWithPath: spec.texturePath) as CFURL, nil),\n'
              '          let thumbCG = CGImageSourceCreateThumbnailAtIndex(\n'
              '            imgSrc, 0, thumbOpts as CFDictionary) else {\n'
              '      // [瞬时快门] 12MP 静照后台落盘,文件可能还没写完 → 解码读空。卡片\n'
              '      // 锚点/几何已建好(贴镜头),只差纹理;每 150ms 重试直到落盘(最多\n'
              '      // 30 次~4.5s)。位姿/几何在 spec 里,重试不动位姿。\n'
              '      let tries = OfficialAetherARKitPlugin.photoCardThumbRetries[name, default: 0]\n'
              '      if tries < 30 {\n'
              '        OfficialAetherARKitPlugin.photoCardThumbRetries[name] = tries + 1\n'
              '        DispatchQueue.main.asyncAfter(deadline: .now() + 0.15) { [weak self, weak node] in\n'
              '          guard let self, let node else { return }\n'
              '          self.renderer(renderer, didAdd: node, for: anchor)\n'
              '        }\n'
              '      } else {\n'
              '        NSLog("[PHOTOCARD] renderer: thumbnail decode FAILED (gave up) for %@", name)\n'
              '        OfficialAetherARKitPlugin.photoCardThumbRetries.removeValue(forKey: name)\n'
              '      }\n'
              '      return\n'
              '    }\n'
              '    OfficialAetherARKitPlugin.photoCardThumbRetries.removeValue(forKey: name)\n'
              '    let image = UIImage(cgImage: thumbCG)   // upright portrait, ~thumb px long edge\n'
              '    NSLog("[PHOTOCARD] renderer building quad for %@ (%d corners) thumb=%dx%d",\n'
              '          name, spec.localCorners.count, thumbCG.width, thumbCG.height)\n'
              '    let c = spec.localCorners\n'
              '    let quadW = CGFloat(simd_length(simd_float3(\n'
              '      c[1].x - c[0].x, c[1].y - c[0].y, c[1].z - c[0].z)))   // TL->TR\n'
              '    let quadH = CGFloat(simd_length(simd_float3(\n'
              '      c[3].x - c[0].x, c[3].y - c[0].y, c[3].z - c[0].z)))   // TL->BL\n'
              '    let texAspect = image.size.height > 0\n'
              '      ? image.size.width / image.size.height : 0.75\n'
              '    let quadAspect = quadH > 0 ? quadW / quadH : 0.46\n'
              '    var u0: CGFloat = 0, u1: CGFloat = 1, v0: CGFloat = 0, v1: CGFloat = 1\n'
              '    if texAspect > quadAspect {            // texture relatively wider → crop width\n'
              '      let f = quadAspect / texAspect; u0 = (1 - f) / 2; u1 = 1 - u0\n'
              '    } else {                               // texture relatively taller → crop height\n'
              '      let f = texAspect / quadAspect; v0 = (1 - f) / 2; v1 = 1 - v0\n'
              '    }\n'
              '    let texUVs = [CGPoint(x: u0, y: v0), CGPoint(x: u1, y: v0),\n'
              '                  CGPoint(x: u1, y: v1), CGPoint(x: u0, y: v1)]   // TL,TR,BR,BL\n'
              '    NSLog("[PHOTOCARD] tex thumb=%.0fx%.0f texAsp=%.3f quadAsp=%.3f",\n'
              '          image.size.width, image.size.height, texAspect, quadAspect)\n'
              '\n'
              '    let positionSource = SCNGeometrySource(vertices: spec.localCorners)\n'
              '    let texSource = SCNGeometrySource(textureCoordinates: texUVs)\n'
              '    let element = SCNGeometryElement(indices: [Int32]([0, 1, 2, 0, 2, 3]),\n'
              '                                     primitiveType: .triangles)\n'
              '    let geometry = SCNGeometry(sources: [positionSource, texSource],\n'
              '                               elements: [element])\n'
              '    let mat = SCNMaterial()\n'
              '    mat.diffuse.contents = image\n'
              '    mat.isDoubleSided = false            // photo on the CAPTURE-FACING side only\n'
              '    mat.cullMode = .front                // = the exact face the double-sided card\n'
              '                                         // showed toward the camera (unchanged view)\n'
              '    mat.lightingModel = .constant       // unlit — show the photo as captured\n'
              '    mat.transparency = 0.7              // RS-style translucent (more see-through)\n'
              '    mat.writesToDepthBuffer = false\n'
              '    mat.diffuse.wrapS = .clamp\n'
              '    mat.diffuse.wrapT = .clamp\n'
              '    geometry.materials = [mat]\n'
              '\n'
              '    // OPAQUE BACK: same quad, rendered only from the AWAY side (cullMode\n'
              '    // .back = the face opposite the photo), so orbiting behind the card shows a\n'
              '    // solid panel instead of the see-through/mirrored photo. Colour follows the\n'
              '    // border\'s four-state rule (black→white/red/yellow via setPhotoCardStates).\n'
              '    let backGeo = SCNGeometry(sources: [positionSource], elements: [element])\n'
              '    let backMat = SCNMaterial()\n'
              '    backMat.diffuse.contents = UIColor.black\n'
              '    backMat.isDoubleSided = false\n'
              '    backMat.cullMode = .back             // the face opposite the photo (away side)\n'
              '    backMat.lightingModel = .constant\n'
              '    backMat.transparency = 1.0           // opaque\n'
              '    backMat.writesToDepthBuffer = false\n'
              '    backGeo.materials = [backMat]\n'
              '\n'
              '    // RS-style FRAME: the four-state border RING around the photo. Starts\n'
              '    // BLACK (= SfM pending); Dart\'s judgement (photo_card_state.dart) flips it\n'
              '    // white (registered) / red (disconnected) / yellow (low parallax) via the\n'
              '    // setPhotoCardStates channel — applied in applyPhotoCardStatesIfDirty.\n'
              '    // Built as a hollow ring (inner edge == photo edge, outer == +12%) so\n'
              '    // it never overlaps the photo (no z-fight, no darkening of the image).\n'
              '    let inner = spec.localCorners\n'
              '    let outer = inner.map { SCNVector3($0.x * 1.12, $0.y * 1.12, $0.z * 1.12) }  // 4× the original 3% — 边框加粗一倍(签决:四态颜色要醒目)\n'
              '    let frameVerts = inner + outer                       // 0-3 inner, 4-7 outer\n'
              '    let frameIdx: [Int32] = [4, 5, 1, 4, 1, 0,           // top edge\n'
              '                             5, 6, 2, 5, 2, 1,           // right edge\n'
              '                             6, 7, 3, 6, 3, 2,           // bottom edge\n'
              '                             7, 4, 0, 7, 0, 3]           // left edge\n'
              '    let frameGeo = SCNGeometry(\n'
              '      sources: [SCNGeometrySource(vertices: frameVerts)],\n'
              '      elements: [SCNGeometryElement(indices: frameIdx, primitiveType: .triangles)])\n'
              '    let frameMat = SCNMaterial()\n'
              '    frameMat.diffuse.contents = UIColor.black\n'
              '    frameMat.isDoubleSided = true\n'
              '    frameMat.lightingModel = .constant\n'
              '    frameMat.transparency = 0.95\n'
              '    frameMat.writesToDepthBuffer = false\n'
              '    frameGeo.materials = [frameMat]\n'
              '\n'
              '    // Wrap border + photo in a CONTAINER we scale per-frame (renderer:updateAtTime)\n'
              '    // for the deliberate distance shrink. Container origin == anchor centroid, so\n'
              '    // scaling shrinks the card toward its own centre without moving it.\n'
              '    let container = SCNNode()\n'
              '    container.addChildNode(SCNNode(geometry: frameGeo))   // border behind/around\n'
              '    container.addChildNode(SCNNode(geometry: backGeo))    // opaque back panel\n'
              '    container.addChildNode(SCNNode(geometry: geometry))   // photo on the front\n'
              '    node.addChildNode(container)\n'
              '    photoCardNodes[name] = container\n'
              '    // 四态边框:登记环+背板材质,并立刻套用 Dart 已推过的状态(卡片\n'
              '    // 节点可能晚于状态到达 —— didAdd 是异步回调)。\n'
              '    photoCardStateMats[name] = [frameMat, backMat]\n'
              '    let initialState = OfficialAetherARKitPlugin.photoCardState(\n'
              '      forPath: spec.evidencePath\n'
              '    )\n'
              '    if initialState != 0 {\n'
              '      let c = Self.photoCardStateColor(initialState)\n'
              '      frameMat.diffuse.contents = c\n'
              '      backMat.diffuse.contents = c\n'
              '    }\n'
              '  }\n'
              '\n'
              '  /// Per-frame: distance-compensated scaling for the floating photo cards +\n'
              '  /// four-state border colour application. Cheap: one distance + scale per\n'
              '  /// card per frame (~百级节点一次 sqrt 可忽略), all on the SceneKit render\n'
              '  /// thread.\n'
              '  func renderer(_ renderer: SCNSceneRenderer, updateAtTime time: TimeInterval) {\n'
              '    OfficialPwNativeTelemetry.shared.noteRenderFrame()  // 遥测 F:FPS 计帧(纳秒级)\n'
              '    updateFeaturePointOverlay(at: time) // stable dynamic LOD; independent of cards\n'
              '    applyPhotoCardStatesIfDirty()  // 四态边框:消费 Dart 推来的状态差量\n'
              '    guard !photoCardNodes.isEmpty, let cam = renderer.pointOfView else { return }\n'
              '    let camPos = cam.simdWorldPosition\n'
              '    for card in photoCardNodes.values {\n'
              '      // RS 复刻显示开关:逐帧幂等套用可见性 —— 隐藏期新建的卡片下一帧即\n'
              '      // 隐藏,重开下一帧全量回显,无需遍历时机协调。\n'
